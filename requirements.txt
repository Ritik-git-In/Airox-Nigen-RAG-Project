# Core app
streamlit>=1.40

# PDF text extraction (keeps page numbers for citations)
pypdf>=5.0

# Vector database (persisted on disk) — also bundles the ONNX MiniLM
# embedding model, so we get local embeddings without PyTorch.
# Pinned to the version verified locally so hosted builds (Streamlit Cloud)
# reproduce the same dependency tree instead of resolving a broken combo.
chromadb==1.5.9

# ONNX runtime powers the lightweight (torch-free) embedding model.
onnxruntime>=1.16

# Kimi / Moonshot is OpenAI-compatible, so we use the OpenAI SDK
openai>=1.50

# Load API keys from a .env file
python-dotenv>=1.0

# OCR fallback for scanned/image-only PDF pages: PyMuPDF renders a page to
# an image (no separate Poppler install needed), pytesseract runs it through
# Tesseract. Tesseract itself is a SEPARATE, non-pip system install — see
# README for the installer link. Without it, OCR is skipped automatically
# (scanned pages are just left out of the index, same as before this feature).
pymupdf>=1.24
pytesseract>=0.3.10

# chromadb -> opentelemetry ships protobuf-generated code. On Streamlit Cloud the
# resolver landed a NEW protobuf next to OLD generated code, which crashes at
# import ("Descriptors cannot be created directly"). Pinning protobuf to 5.x keeps
# the runtime AND the generated code on the modern descriptor style, and is
# accepted by both streamlit (protobuf<8,>=3.20) and opentelemetry-proto (>=5,<7).
# 6.x is the version verified locally.
protobuf>=5,<7
